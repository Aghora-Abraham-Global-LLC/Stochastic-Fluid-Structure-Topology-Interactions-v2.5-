In the foundational Version 1.0 monograph (\texttt{DOI: 10.5281/zenodo.22806453}) \cite{ghulame2026spde_v1}, initial multiscale reaction-diffusion formulations were posed on deformable manifolds. \begin{enumerate}
  \item \textbf{Generalized It\^o--Wentzell Pullback on $\operatorname{Diff}(\mathcal{M}_0)$:} Pulling back an SPDE from an evolving manifold $\mathcal{M}_t = \Phi_t(\mathcal{M}_0)$ to a static compact reference manifold $\mathcal{M}_0$ via a Stratonovich Kunita flow $\Phi_t$ generates non-trivial second-order Lie derivative advection $\frac{1}{2}\sum_{i=1}^N \mathcal{L}_{\mathbf{v}_i}\mathcal{L}_{\mathbf{v}_i}\rho$ and stochastic diffusion cross-variation $\sum_{i=1}^N \rho_{WB}^i \mathcal{L}_{\mathbf{v}_i}(\Phi_t^*\tilde{\mathbf{G}})$.
  \item \textbf{Mean-Zero Fractional Gelfand Triples:} Because constant functions lie in the nullspace of the closed manifold Laplace--Beltrami operator $-\Delta_{g(t)}$, strict coercivity requires projecting onto the mean-zero subspace $\dot{L}^2(\mathcal{M}_0, g(t))$ and establishing a fractional Poincaré--Lichnerowicz inequality via Bochner spectral subordination.
  \item \textbf{Orthonormal Frame Bundle $O(\mathcal{M}_0)$ Homogenization:} Naive two-scale homogenization in local manifold coordinates introduces $\mathcal{O}(\epsilon^2 \|\mathrm{Riem}_{g(t)}\|)$ curvature corruption. Lifting the cell problem to horizontal vector fields on $O(\mathcal{M}_0)$ enforces $\Gamma_{\beta\gamma}^\alpha(0) = 0$ at the normal coordinate pole, annihilating the curvature defect.

  \item \textbf{Chebyshev--Lanczos Krylov Subspace IMEX Discretization:} Because spatial Fourier transforms (FFTs) do not diagonalize spatially varying Riemannian metrics $g_{ij}(t,x)$, fractional powers $(-\Delta_{g(t)})^s$ are evaluated via matrix-free Lanczos tridiagonal Krylov projections \cite{saad1992analysis, higham2008functions}.

\section{Kunita Stochastic Flows on $\operatorname{Diff}(\mathcal{M}_0)$ and the Generalized It\^o--Wentzell Pullback}

Let $(\Omega, \mathcal{F}, (\mathcal{F}_t)_{t \ge 0}, \mathbb{P})$ be a complete filtered probability space supporting two Cylindrical Wiener processes: a state-space driving noise $W_t$ and a manifold-deforming turbulence noise $B_t = (B_t^1, \dots, B_t^N)$. Let $(\mathcal{M}_0, g_0)$ be a compact, orientable, connected $d$-dimensional smooth Riemannian reference manifold without boundary ($\partial\mathcal{M}_0 = \emptyset$).

\begin{definition}[Volume-Preserving Solenoidal Kunita Stochastic Flow]
The physical domain $\mathcal{M}_t = \Phi_t(\mathcal{M}_0)$ evolves via a stochastic flow of diffeomorphisms $\Phi_t : \mathcal{M}_0 \to \mathcal{M}_t$ satisfying the Stratonovich stochastic differential equation on $\operatorname{Diff}(\mathcal{M}_0)$ \cite{kunita1990stochastic, baxendale1984brownian}:
\begin{equation}\label{eq:kunita_flow}
  \mathrm{d}\Phi_t(x) = \mathbf{u}(t, \Phi_t(x))\,\mathrm{d}t + \sum_{i=1}^N \mathbf{v}_i(\Phi_t(x)) \circ \mathrm{d}B_t^i, \qquad \Phi_0(x) = x \in \mathcal{M}_0,
\end{equation}
where the mean drift $\mathbf{u}(t,\cdot) \in \Gamma(T\mathcal{M}_t)$ and the stochastic wind-shear modes $\mathbf{v}_i \in \Gamma(T\mathcal{M}_t)$ are smooth, divergence-free (solenoidal) vector fields with respect to the Riemannian volume form $\mathrm{d}\mu_{g(t)}$:
\begin{equation}\label{eq:solenoidal_cond}
  \operatorname{div}_{g(t)} \mathbf{u}(t, \cdot) = 0, \qquad \operatorname{div}_{g(t)} \mathbf{v}_i = 0 \quad \forall i \in \{1, \dots, N\}.
\end{equation}
\end{definition}

Let $\tilde{\rho}(t, y)$ denote the physical scalar field (e.g., atmospheric dust-plasma density or navigation probability density) evolving on the moving manifold $\mathcal{M}_t$ according to the Eulerian SPDE $\mathrm{d}\tilde{\rho}(t,y) = \tilde{A}(t,\tilde{\rho}(t,y))\,\mathrm{d}t + \tilde{\mathbf{G}}(t,\tilde{\rho}(t,y))\,\mathrm{d}W_t$. Pulling back the metric $g(t) \coloneqq \Phi_t^* \tilde{g}(t)$ and the scalar density $\rho(t, x) \coloneqq (\Phi_t^* \tilde{\rho})(t, x) = \tilde{\rho}(t, \Phi_t(x))$ to the static reference manifold $\mathcal{M}_0$ yields the generalized It\^o--Wentzell pullback equation.

\begin{theorem}[Generalized It\^o--Wentzell Pullback on Evolving Manifolds \cite{ananda2026stochastic_fluid_v2, krylov2011ito}]\label{thm:ito_wentzell}
Assume $\mathbf{u}, \mathbf{v}_i \in C_b^3(T\mathcal{M}_t)$ satisfy the solenoidal condition \eqref{eq:solenoidal_cond}, and let the quadratic cross-variation between the state noise $W_t$ and the Kunita flow noise $B_t^i$ be parameterized by $\mathrm{d}\langle W, B^i \rangle_t = \rho_{WB}^i\,\mathrm{d}t$ with $|\rho_{WB}^i| \le 1$. Then the pullback field $\rho(t) = \Phi_t^*\tilde{\rho}(t)$ on $\mathcal{M}_0$ satisfies the It\^o SPDE:
\begin{align}\label{eq:ito_wentzell_pullback}
  \mathrm{d}\rho(t) &= \left( \Phi_t^* \tilde{A}(t, \tilde{\rho}) + \mathcal{L}_{\mathbf{u}} \rho + \frac{1}{2} \sum_{i=1}^N \mathcal{L}_{\mathbf{v}_i} \mathcal{L}_{\mathbf{v}_i} \rho + \sum_{i=1}^N \rho_{WB}^i \mathcal{L}_{\mathbf{v}_i} \left(\Phi_t^* \tilde{\mathbf{G}}(t, \tilde{\rho})\right) \right) \mathrm{d}t \nonumber \\
  &\quad + \Phi_t^* \tilde{\mathbf{G}}(t, \tilde{\rho})\,\mathrm{d}W_t + \sum_{i=1}^N \mathcal{L}_{\mathbf{v}_i} \rho \,\mathrm{d}B_t^i,
\end{align}
where $\mathcal{L}_{\mathbf{v}_i}\rho = \mathbf{v}_i \cdot \nabla_{g(t)}\rho$ denotes the Lie derivative along $\mathbf{v}_i$.
\end{theorem}

\begin{remark}[Skew-Symmetry of Solenoidal Lie Derivatives]\label{rem:skew_symmetry}
Because the vector fields $\mathbf{v}_i$ are strictly solenoidal ($\operatorname{div}_{g(t)} \mathbf{v}_i = 0$), the Lie derivative operator $\mathcal{L}_{\mathbf{v}_i}$ is strictly skew-symmetric on the mean-zero Hilbert space $\dot{H}_t$:
\begin{align}\label{eq:skew_symmetry}
  \langle \mathcal{L}_{\mathbf{v}_i} \rho, \rho \rangle_{\dot{H}_t}
  &= \int_{\mathcal{M}_0} \rho (\mathbf{v}_i \cdot \nabla_{g(t)} \rho) \,\mathrm{d}\mu_{g(t)}
   = \frac{1}{2} \int_{\mathcal{M}_0} \mathbf{v}_i \cdot \nabla_{g(t)} (\rho^2) \,\mathrm{d}\mu_{g(t)} \nonumber \\
  &= -\frac{1}{2} \int_{\mathcal{M}_0} \rho^2 \operatorname{div}_{g(t)}(\mathbf{v}_i) \,\mathrm{d}\mu_{g(t)} = 0.
\end{align}
This differential identity theoretically guarantees that the advective Lie derivatives do not destroy $L^2$ stability, rendering the energy dissipativity bounds in Theorem \ref{thm:master_stability} mathematically transparent.
\end{remark}

\begin{proposition}[Regularity of the Pullback Metric Time-Derivative]\label{prop:metric_trace_bound}
Under the smooth Kunita flow \eqref{eq:kunita_flow}, the pullback Riemannian metric $g(t) = \Phi_t^*\tilde{g}(t)$ satisfies the uniform trace velocity bound:
\begin{equation}\label{eq:metric_cg_bound}
  C_g \coloneqq \sup_{t \in [0,T]} \left\| \operatorname{Tr}_{g(t)}\!\left(\partial_t g(t)\right) \right\|_{L^\infty(\mathcal{M}_0)} < \infty.
\end{equation}
\end{proposition}

\begin{remark}[Independence Specialization, Remark 2.3 of v2.0]\label{rem:independence_spec}
When the internal reaction-diffusion noise $W_t$ and the environmental Kunita turbulence $B_t^i$ are mutually independent ($\rho_{WB}^i \equiv 0$), the cross-variation commutator $\sum_{i=1}^N \rho_{WB}^i \mathcal{L}_{\mathbf{v}_i}(\Phi_t^*\tilde{\mathbf{G}})$ vanishes identically. In planetary EDL atmospheric entry (Section 8), when radar altimeter scintillation ($W_t$) is physically correlated with dust-devil shear ($B_t^i$), $\rho_{WB} \in (0, 1]$ captures the exact It\^o--Wentzell drift correction.
\end{remark}

\section{Mean-Zero Fractional Gelfand Triples and Bochner Spectral Subordination}

To eliminate the trivial constant nullspace of the Laplace--Beltrami operator $-\Delta_{g(t)}$ on the compact boundaryless manifold $(\mathcal{M}_0, g(t))$, we restrict all functional spaces to the mean-zero subspace:
\begin{equation}\label{eq:mean_zero_hilbert}
  \dot{H}_t \coloneqq \left\{ v \in L^2(\mathcal{M}_0, \mathrm{d}\mu_{g(t)}) \;\middle|\; \int_{\mathcal{M}_0} v(x)\,\mathrm{d}\mu_{g(t)}(x) = 0 \right\}.
\end{equation}
Because the Kunita vector fields $\mathbf{u}, \mathbf{v}_i$ are strictly solenoidal \eqref{eq:solenoidal_cond}, Stokes' theorem guarantees $\int_{\mathcal{M}_0} (\mathcal{L}_{\mathbf{v}_i}\rho)\,\mathrm{d}\mu_{g(t)} = -\int_{\mathcal{M}_0} \rho \operatorname{div}_{g(t)}\mathbf{v}_i\,\mathrm{d}\mu_{g(t)} = 0$, leaving $\dot{H}_t$ invariant under the stochastic flow.

\begin{definition}[Bochner Spectral Subordination of the Fractional Laplacian \cite{caffarelli2007extension, kwasnicki2017ten}]
For any fractional order $s \in (0, 1)$, the non-local fractional Laplace--Beltrami operator $(-\Delta_{g(t)})^s : \dot{V}_t^s \to (\dot{V}_t^s)^*$ is defined via Bochner's subordination integral over the heat semigroup $\mathrm{e}^{\tau \Delta_{g(t)}}$. Using the Gamma function identity $\Gamma(-s) = -\frac{\Gamma(1-s)}{s}$, we express the integral to make positive definiteness explicit for positive functions:
\begin{equation}\label{eq:bochner_subordination}
  (-\Delta_{g(t)})^s v(x) = \frac{s}{\Gamma(1-s)} \int_0^\infty \left( v(x) - \mathrm{e}^{\tau \Delta_{g(t)}} v(x) \right) \frac{\mathrm{d}\tau}{\tau^{1+s}} = \sum_{k=1}^\infty \lambda_k(t)^s \langle v, e_k(t) \rangle_{\dot{H}_t} e_k(t),
\end{equation}
generating the mean-zero fractional Gelfand triple $\dot{V}_t^s \coloneqq \dot{H}^s(\mathcal{M}_0, g(t)) \hookrightarrow \dot{H}_t \hookrightarrow (\dot{V}_t^s)^*$.
\end{definition}

\begin{proposition}[Lichnerowicz Fractional Poincaré Spectral Gap \cite{lichnerowicz1958geometrie, ananda2026stochastic_fluid_v2}]\label{prop:lichnerowicz_gap}
Suppose the evolving Riemannian manifold $(\mathcal{M}_0, g(t))$ has Ricci curvature uniformly bounded below by $\operatorname{Ric}_{g(t)} \ge K\,g(t)$ with $K > 0$ and dimension $d \ge 2$. By the Lichnerowicz theorem, the first non-zero eigenvalue satisfies $\lambda_1(t) \ge \frac{d}{d-1}K$, and every $v \in \dot{V}_t^s$ obeys the sharp fractional Poincaré inequality:
\begin{equation}\label{eq:fractional_poincare}
  \|v\|_{\dot{V}_t^s}^2 = \left\langle (-\Delta_{g(t)})^s v, v \right\rangle_{\dot{H}_t} \ge \mathcal{C}_s^* \|v\|_{\dot{H}_t}^2, \qquad \mathcal{C}_s^* \coloneqq \left( \frac{d\,K}{d-1} \right)^s > 0.
\end{equation}
\end{proposition}

\section{Two-Scale Homogenization on the Orthonormal Frame Bundle $O(\mathcal{M}_0)$}

Consider a rapidly oscillating micro-scale conductivity/diffusion tensor $\mathbf{D}_\epsilon(x) = \mathbf{D}(x, x/\epsilon)$ with period cell $Y = \mathbb{T}^d$ as $\epsilon \to 0$ \cite{bensoussan2011asymptotic, jikov2012homogenization}. In naive local coordinates on a curved manifold, two-scale asymptotic expansion $u_\epsilon(x) = u_0(x) + \epsilon u_1(x, y) + \epsilon^2 u_2(x, y)$ mixes the macroscopic Christoffel symbols $\Gamma_{\beta\gamma}^\alpha(x)$ with the microscopic cell derivatives $\partial_{y^\alpha}$, inducing an artificial $\mathcal{O}(\epsilon^2 \|\mathrm{Riem}_{g(t)}\|)$ curvature defect.

\begin{theorem}[Frame Bundle Horizontal Homogenization and $\mathcal{O}(\epsilon^2)$ Curvature Annihilation \cite{ananda2026stochastic_fluid_v2, hsu2002stochastic}]\label{thm:frame_bundle_homog}
Lift the two-scale expansion to the orthonormal frame bundle $\pi : O(\mathcal{M}_0) \to \mathcal{M}_0$ equipped with the Levi-Civita connection $\omega$ and canonical horizontal vector fields $H_1, \dots, H_d$. In horizontal normal coordinates centered at the fiber pole $y = 0$, the connection coefficients vanish identically:
\begin{equation}\label{eq:christoffel_pole_zero}
  \Gamma_{\beta\gamma}^\alpha(0) = 0, \qquad \partial_\mu \Gamma_{\beta\gamma}^\alpha(0) + \partial_\beta \Gamma_{\gamma\mu}^\alpha(0) + \partial_\gamma \Gamma_{\mu\beta}^\alpha(0) = 0.
\end{equation}

\end{align}

\end{theorem}

__global__ void spde_imex_fractional_krylov_kernel(
    const double* __restrict__ rho_in, 
    double* __restrict__ rhs_out,
    const double* __restrict__ u_tda_lasry_lions, // C^{1,1} proximal grad

    double drift = -advection[idx] - (kappa * rho_in[idx]) - u_tda_lasry_lions[idx];

    rhs_out[idx] = rho_in[idx] + drift * dt + diffusion;
}
